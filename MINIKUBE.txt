kubectl create deployment mynginx --image=nginx

kubectl get deployments

kubectl expose deployment mynginx --type=NodePort --port=80 --target-port=80

kubectl scale deployment mynginx --replicas=4

kubectl get service myngnix

kubectl port-forward svc/mynginx 8081:80

Open localhost after that delete

kubectl delete deployment mynginx

pipeline 
pipeline {
    agent any
    tools{
        maven 'maven'
    }
    stages {
        stage('git repo & clean') {
            steps {
                bat "rmdir /s /q jenkin-web || exit 0"
                bat "git clone https://github.com/Hasinireddy11/jenkin-web.git"
                bat "mvn clean -f jenkin-web"
            }
        }
        stage('install') {
            steps {
                bat "mvn install -f jenkin-web"
            }
        }
        stage('test') {
            steps {
                bat "mvn test -f jenkin-web"
            }
        }
        stage('package') {
            steps {
                bat "mvn package -f jenkin-web"
            }
        }
    }
}
